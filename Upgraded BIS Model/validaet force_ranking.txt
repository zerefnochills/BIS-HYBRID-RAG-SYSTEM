# Validate the key logic changes are all correct
import re

# Test _force_rank_is_match logic
def force_rank(candidates, is_numbers):
    if not is_numbers or not candidates:
        return candidates
    matched, unmatched = [], []
    for item in candidates:
        _, sid = item
        if any(re.search(rf'\bIS\s*{re.escape(n)}\b', sid, re.IGNORECASE) for n in is_numbers):
            matched.append(item)
        else:
            unmatched.append(item)
    return matched + unmatched

# Test: oracle correctly places IS 2185 (Part 2) before IS 2185 (Part 3)
